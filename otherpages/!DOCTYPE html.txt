  <script>

    const heightInput = document.getElementById("height");
    const confirmButton = document.getElementById("confirmButton");
    const result = document.getElementById("result");
    const calculator = document.getElementById("calculator");


    function calculateHeight() {

      const height = parseFloat(heightInput.value);

      // Do nothing if input is empty or invalid
      if (isNaN(height) || height <= 0) {
        result.textContent = "Please enter a valid height.";
        return;
      }

      // Divide height by 7
      const answer = height / 7;

      // Show result
      result.textContent = answer.toFixed(2);


      // Restart bounce animation
      calculator.classList.remove("bounce");

      // Force browser to restart animation
      void calculator.offsetWidth;

      calculator.classList.add("bounce");
    }


    // Confirm button
    confirmButton.addEventListener(
      "click",
      calculateHeight
    );


    // Allow pressing Enter
    heightInput.addEventListener(
      "keydown",
      function(event) {

        if (event.key === "Enter") {
          calculateHeight();
        }

      }
    );

  </script>
